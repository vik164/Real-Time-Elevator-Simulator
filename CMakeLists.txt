cmake_minimum_required(VERSION 3.16)
project(RealTimeElevatorSimulator VERSION 1.0 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
find_package(Threads REQUIRED)
add_library(elevator_core src/model.cpp src/udp.cpp src/simulator.cpp)
target_include_directories(elevator_core PUBLIC include)
target_link_libraries(elevator_core PUBLIC Threads::Threads)
if(WIN32)
  target_link_libraries(elevator_core PUBLIC ws2_32)
endif()
if(MSVC)
  target_compile_options(elevator_core PRIVATE /W4 /permissive-)
else()
  target_compile_options(elevator_core PRIVATE -Wall -Wextra -Wpedantic)
endif()
add_executable(elevator_simulator src/main.cpp)
target_link_libraries(elevator_simulator PRIVATE elevator_core)
include(CTest)
if(BUILD_TESTING)
  add_executable(elevator_tests tests/tests.cpp)
  target_link_libraries(elevator_tests PRIVATE elevator_core)
  add_test(NAME elevator_tests COMMAND elevator_tests ${CMAKE_CURRENT_SOURCE_DIR}/resources)
  set_tests_properties(elevator_tests PROPERTIES TIMEOUT 30)
endif()
